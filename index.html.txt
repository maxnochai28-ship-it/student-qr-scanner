<script>
let scanner=null;
let running=false;
let lastCode="";
let lastTime=0;

async function startScan(){
  if(running)return;
  if(!window.Html5Qrcode){
    document.getElementById("result").textContent=
      "โหลดระบบสแกนไม่สำเร็จ กรุณาตรวจสอบอินเทอร์เน็ต";
    return;
  }
  try{
    scanner=new Html5Qrcode("reader");
    const cameras=await Html5Qrcode.getCameras();
    if(!cameras.length)throw Error("ไม่พบกล้อง");
    const camera=cameras.find(c=>/back|rear|environment/i.test(c.label))
      ||cameras[0];
    await scanner.start(
      camera.id,
      {fps:10,qrbox:{width:240,height:240}},
      text=>{
        const now=Date.now();
        if(text===lastCode&&now-lastTime<3000)return;
        lastCode=text;
        lastTime=now;
        document.getElementById("result").textContent=
          "อ่าน QR Code สำเร็จ: "+text;
      },
      ()=>{}
    );
    running=true;
    document.getElementById("result").textContent=
      "กล้องพร้อมใช้งาน กรุณาส่อง QR Code";
  }catch(e){
    document.getElementById("result").textContent=
      "เปิดกล้องไม่ได้: "+e.message+
      " — โปรดอนุญาตให้เว็บไซต์ใช้กล้อง";
    if(scanner){try{await scanner.clear()}catch(_){}scanner=null}
  }
}

async function stopScan(){
  if(scanner){
    try{if(running)await scanner.stop()}catch(_){}
    try{await scanner.clear()}catch(_){}
  }
  scanner=null;
  running=false;
  document.getElementById("result").textContent="หยุดกล้องแล้ว";
}
</script>
